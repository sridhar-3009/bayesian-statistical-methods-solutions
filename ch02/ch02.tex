\section*{A five-minute primer on the tools of this chapter}
\addcontentsline{toc}{section}{Primer: the tools of this chapter}

\begin{recall}
\textbf{1. Conjugate pairs.} A prior is \emph{conjugate} to a likelihood if the posterior is in the same family as the prior. The ones used in this chapter:
\begin{center}
\small
\begin{tabular}{lll}\toprule
Likelihood & Prior & Posterior\\\midrule
$Y\sim\dist{Binomial}(n,\theta)$ & $\dist{Beta}(a,b)$ & $\dist{Beta}(a+Y,\ b+n-Y)$\\
$Y_i\iid\dist{Poisson}(\lambda)$ & $\dist{Gamma}(a,b)$ & $\dist{Gamma}(a+\sum Y_i,\ b+n)$\\
$Y_i\iid\dist{N}(\mu,\sigma^2)$, $\sigma^2$ known & $\dist N(m,\sigma^2/k)$ & $\dist N\big(\frac{n\bar Y+km}{n+k},\frac{\sigma^2}{n+k}\big)$\\
$Y_i\iid\dist{N}(0,\sigma^2)$ & $\dist{InvGamma}(a,b)$ & $\dist{InvGamma}(a+\frac n2,\ b+\frac{\sum Y_i^2}2)$\\\bottomrule
\end{tabular}

\smallskip
Reading the updates: a $\dist{Beta}(a,b)$ prior acts like $a$ earlier successes and $b$ failures; a $\dist{Gamma}(a,b)$ prior like $a$ events in $b$ units of exposure; a $\dist N(m,\sigma^2/k)$ prior like $k$ extra observations equal to $m$.
\end{center}

\medskip
\textbf{2. Recognising a posterior.} Multiply likelihood $\times$ prior, drop every factor without the parameter, collect powers and exponents, and match a kernel:
$\theta^{a-1}(1-\theta)^{b-1}\to$ Beta; \ $\lambda^{a-1}e^{-b\lambda}\to$ Gamma; \ $x^{-a-1}e^{-b/x}\to$ InvGamma; \ $e^{-\frac{A}{2}x^2+Bx}\to\dist N(B/A,1/A)$.

\medskip
\textbf{3. Improper priors.} A prior with $\int\pi(\theta)\dd\theta=\infty$ (e.g.\ $\pi(\theta)\propto1$) is allowed \emph{only if} the posterior is proper, i.e.\ $\int f(Y\mid\theta)\pi(\theta)\dd\theta<\infty$.

\medskip
\textbf{4. Jeffreys prior.} $\pi(\theta)\propto\sqrt{I(\theta)}$ with Fisher information $I(\theta)=-\E\big[\frac{\dd^2}{\dd\theta^2}\log f(Y\mid\theta)\big]$. It gives the same answer whichever parameterisation you start from.

\medskip
\textbf{5. Mixture priors.} If $\pi(\theta)=\sum_kw_k\pi_k(\theta)$ with each $\pi_k$ conjugate, the posterior is $\sum_kw_k'p_k(\theta\mid Y)$ with new weights $w_k'\propto w_k\,m_k(Y)$, where $m_k(Y)=\int f(Y\mid\theta)\pi_k(\theta)\dd\theta$ is the marginal likelihood under component $k$.

\medskip
\textbf{6. Empirical Bayes.} Estimate the prior's hyperparameters from the data (e.g.\ by matching moments), then do an ordinary Bayesian analysis with that prior.
\end{recall}

\exercise{2.1 --- Frequentist coverage of a Bayesian interval}

\begin{problem}{1}
Assume $Y_1,\dots,Y_n|\mu\iid\dist{Normal}(\mu,\sigma^2)$ where $\sigma^2$ is fixed and the unknown mean $\mu$ has prior $\mu\sim\dist{Normal}(0,\sigma^2/m)$.
\begin{enumerate}[(a)]
\item Give a 95\% posterior interval for $\mu$.
\item Select a value of $m$, and argue that for this choice your 95\% posterior credible interval has frequentist coverage 0.95. That is, if you draw many samples of size $n$ and compute the 95\% interval following the formula in (a) for each sample, in the long-run 95\% of the intervals will contain the true value of $\mu$.
\end{enumerate}
\end{problem}

\begin{restate}
Work out the posterior of a normal mean under a normal prior, write its 95\% credible interval, then find the prior strength $m$ for which this Bayesian interval also behaves like a classical 95\% confidence interval.
\end{restate}

\begin{recall}
\begin{itemize}
\item \textbf{Precision} $=1/\text{variance}$. For normal data with a normal prior, precisions add and the posterior mean is a precision-weighted average of the data mean and the prior mean.
\item $\bar Y\sim\dist N(\mu,\sigma^2/n)$, so $\bar Y$ carries precision $n/\sigma^2$.
\item For $Z\sim\dist N(0,1)$: $\Prob(|Z|\le1.96)=0.95$.
\end{itemize}
\end{recall}

\begin{keyidea}
The prior $\dist N(0,\sigma^2/m)$ is worth $m$ imaginary observations equal to 0. As $m\to0$ the prior disappears and the Bayesian interval becomes the classical one.
\end{keyidea}

\solution
\step{(a) Step 1 --- likelihood and prior as functions of $\mu$.}
\[
f(\mathbf Y\mid\mu)\propto\exp\Big\{-\frac{n}{2\sigma^2}(\mu-\bar Y)^2\Big\},\qquad\pi(\mu)\propto\exp\Big\{-\frac{m}{2\sigma^2}\mu^2\Big\}.
\]
\why{$\sum_i(Y_i-\mu)^2=\sum_i(Y_i-\bar Y)^2+n(\bar Y-\mu)^2$, and the first sum does not involve $\mu$.}

\step{(a) Step 2 --- multiply and complete the square.}
The exponent is $-\frac1{2\sigma^2}\big[n(\mu-\bar Y)^2+m\mu^2\big]$. Expanding, the $\mu^2$ coefficient is $\frac{n+m}{\sigma^2}$ and the $\mu$ coefficient is $\frac{n\bar Y}{\sigma^2}$, so (Recall box of the primer, point 2)
\[
\mu\mid\mathbf Y\sim\dist{Normal}\!\left(\frac{n\bar Y}{n+m},\ \frac{\sigma^2}{n+m}\right).
\]

\step{(a) Step 3 --- the interval.}
For a normal posterior, mean $\pm1.96\times$SD contains 95\% (equal-tailed = HPD by symmetry):
\[
\frac{n\bar Y}{n+m}\ \pm\ 1.96\,\frac{\sigma}{\sqrt{n+m}}.
\]

\step{(b) Choose $m=0$.}
The interval becomes $\bar Y\pm1.96\,\sigma/\sqrt n$. Now think of repeated samples with the true $\mu$ fixed: $\bar Y\sim\dist N(\mu,\sigma^2/n)$, so $Z=\frac{\bar Y-\mu}{\sigma/\sqrt n}\sim\dist N(0,1)$ and
\[
\Prob\Big(\mu\in\bar Y\pm1.96\tfrac{\sigma}{\sqrt n}\Big)=\Prob\Big(|\bar Y-\mu|\le1.96\tfrac{\sigma}{\sqrt n}\Big)=\Prob(|Z|\le1.96)=0.95,
\]
for \emph{every} value of $\mu$. ($m=0$ is the limit of an infinitely vague prior --- the flat prior $\pi(\mu)\propto1$.)

\step{Why other $m$ do not work.}
For general $m$, $\frac{n\bar Y}{n+m}-\mu$ has mean $-\frac{m\mu}{n+m}$ (a bias toward 0) and SD $\frac{\sqrt n\sigma}{n+m}$. Standardising gives the coverage
\[
C(\mu)=\Phi\!\Big(1.96\sqrt{\tfrac{n+m}{n}}+\tfrac{m\mu}{\sqrt n\,\sigma}\Big)-\Phi\!\Big(-1.96\sqrt{\tfrac{n+m}{n}}+\tfrac{m\mu}{\sqrt n\,\sigma}\Big).
\]
For $n=10$, $\sigma=1$:
\begin{center}\includegraphics[width=0.72\linewidth]{ex01_coverage.pdf}\end{center}
Near $\mu=0$ the coverage exceeds 0.95 (the prior is right); far from 0 it collapses (the interval is pulled toward the wrong place).

\begin{sanity}
Simulation with $m=0$, $\mu=1.7$, $n=10$ and $10^5$ repeated samples: 94.94\% of the intervals contain $\mu$. \checkmark
\end{sanity}

\begin{answer}
(a) $\displaystyle \frac{n\bar Y}{n+m}\pm1.96\frac{\sigma}{\sqrt{n+m}}$. \qquad
(b) $m=0$: the interval becomes $\bar Y\pm1.96\sigma/\sqrt n$, whose coverage is exactly 0.95 for every $\mu$.
\end{answer}

\begin{takeaway}
A credible interval is a statement about $\mu$ \emph{given the data}; a confidence interval is a statement about the \emph{procedure}. They coincide under a flat prior here.
\end{takeaway}

\exercise{2.2 --- Is Reggie Jackson really ``Mr.\ October''?}

\begin{problem}{2}
The Major League Baseball player Reggie Jackson is known as ``Mr. October'' for his outstanding performances in the World Series (which takes place in October). Over his long career he played in 2820 regular-season games and hit 563 home runs in these games (a player can hit 0, 1, 2, \dots{} home runs in a game). He also played in 27 World Series games and hit 10 home runs in these games. Assuming uninformative conjugate priors, summarize the posterior distribution of his home-run rate in the regular season and World Series. Is there sufficient evidence to claim that he performs better in the World Series?
\end{problem}

\begin{restate}
Estimate his home runs per game in the regular season and in the World Series, and judge whether the World Series rate is convincingly higher.
\end{restate}

\begin{recall}
Poisson--gamma: if $Y_1,\dots,Y_n\iid\dist{Poisson}(\lambda)$ and $\lambda\sim\dist{Gamma}(a,b)$, then $\lambda\mid\mathbf Y\sim\dist{Gamma}(a+\sum Y_i,\ b+n)$. Only the total count and the number of games are needed.
\end{recall}

\begin{keyidea}
Fit one Poisson--gamma model per setting; compare the two rates by drawing from both posteriors and computing their ratio (Monte Carlo).
\end{keyidea}

\solution
\step{Step 1 --- prior.} $\dist{Gamma}(0.01,0.01)$ for each rate: prior mean 1, variance 100 --- worth only 0.01 games of data, so it is essentially uninformative.
\step{Step 2 --- posteriors.}
Regular season: $\dist{Gamma}(0.01+563,\ 0.01+2820)$. World Series: $\dist{Gamma}(0.01+10,\ 0.01+27)$.
\begin{center}
\begin{tabular}{lcccc}\toprule
 & Posterior & Mean $=a/b$ & SD $=\sqrt a/b$ & 95\% interval\\\midrule
Regular season & $\dist{Gamma}(563.01,\ 2820.01)$ & 0.200 & 0.008 & (0.184, 0.217)\\
World Series & $\dist{Gamma}(10.01,\ 27.01)$ & 0.371 & 0.117 & (0.178, 0.633)\\\bottomrule
\end{tabular}
\end{center}
\begin{center}\includegraphics[width=0.7\linewidth]{ex02_reggie.pdf}\end{center}
\step{Step 3 --- compare by simulation.}
Draw $10^6$ values of each rate, then count how often $\lambda_{WS}>\lambda_R$ and summarise $\lambda_{WS}/\lambda_R$:
\[
\Prob(\lambda_{WS}>\lambda_{R}\mid\text{data})\approx0.951,\qquad
\frac{\lambda_{WS}}{\lambda_R}:\ \text{median }1.80,\ \text{95\% interval }(0.89,\ 3.19).
\]

\begin{sanity}
Posterior means are essentially the raw rates: $563/2820=0.200$ and $10/27=0.370$. \checkmark
\end{sanity}

\begin{answer}
Regular season $\approx0.200$ HR/game (0.184--0.217); World Series $\approx0.371$ HR/game (0.178--0.633). There is a 95\% posterior probability that his World Series rate is higher, but the interval for the ratio includes 1: the evidence is \textbf{suggestive but not conclusive} (27 games is a small sample).
\end{answer}

\begin{takeaway}
Beware selection bias: he earned the nickname \emph{because} of a few spectacular games, so testing the nickname on the same data overstates the evidence.
\end{takeaway}

\exercise{2.3 --- Conjugate prior for the exponential rate}

\begin{problem}{3}
Assume that $Y|\theta\sim\dist{Exponential}(\theta)$, i.e., $Y|\theta\sim\dist{Gamma}(1,\theta)$. Find a conjugate prior distribution for $\theta$, and derive the resulting posterior distribution.
\end{problem}

\begin{restate}
Find a family of priors such that, after observing an exponential waiting time, the posterior stays in the same family.
\end{restate}

\begin{recall}
$\dist{Exponential}(\theta)$ with rate $\theta$: $f(y\mid\theta)=\theta e^{-\theta y}$ for $y>0$, mean $1/\theta$. Gamma kernel: $\theta^{a-1}e^{-b\theta}$.
\end{recall}

\begin{keyidea}
Look at the likelihood \emph{as a function of $\theta$}: $\theta^1e^{-y\theta}$ has the Gamma shape ``power of $\theta$ times exponential of $\theta$''.
\end{keyidea}

\solution
\step{Step 1 --- likelihood as a function of $\theta$.} $f(Y\mid\theta)=\theta\,e^{-\theta Y}$.
\step{Step 2 --- try a Gamma prior.} $\pi(\theta)\propto\theta^{a-1}e^{-b\theta}$.
\step{Step 3 --- multiply.}
\[
p(\theta\mid Y)\propto\theta e^{-\theta Y}\cdot\theta^{a-1}e^{-b\theta}=\theta^{(a+1)-1}\,e^{-(b+Y)\theta}.
\]
\why{Add the powers of $\theta$: $1+(a-1)=a=(a+1)-1$; add the exponents: $-\theta Y-b\theta=-(b+Y)\theta$.}
This is the $\dist{Gamma}(a+1,\ b+Y)$ kernel, so the Gamma family is conjugate.
\step{Step 4 --- $n$ observations.} The likelihood becomes $\theta^ne^{-\theta\sum Y_i}$, giving $\theta\mid\mathbf Y\sim\dist{Gamma}(a+n,\ b+\sum Y_i)$.

\begin{sanity}
Posterior mean $\frac{a+n}{b+\sum Y_i}$. With a vague prior ($a,b\to0$) this becomes $n/\sum Y_i=1/\bar Y$, the usual estimate of a rate. \checkmark
\end{sanity}

\begin{answer}
The Gamma family is conjugate: $\theta\sim\dist{Gamma}(a,b)\Rightarrow\theta\mid Y\sim\dist{Gamma}(a+1,\,b+Y)$; for $n$ observations, $\dist{Gamma}(a+n,\,b+\sum Y_i)$.
\end{answer}

\begin{takeaway}
Interpretation: $a$ = prior number of events, $b$ = prior total waiting time. The posterior mean is ``events $\div$ total time''.
\end{takeaway}

\exercise{2.4 --- Factory injury rate}

\begin{problem}{4}
A large factory's objective is to keep the injury rate below 4 injuries per month. In the past 6 months there have been 4, 5, 2, 7, 6 and 6 injuries. Conduct a Bayesian analysis of these data and compute the posterior probability that the injury rate exceeds four injuries per month.
\end{problem}

\begin{restate}
From six monthly injury counts, estimate the monthly injury rate and how likely it is to exceed the target of 4.
\end{restate}

\begin{recall}
Poisson--gamma (primer, point 1): $\lambda\mid\mathbf Y\sim\dist{Gamma}(a+\sum Y_t,\ b+n)$. A tail probability $\Prob(\lambda>4)$ is \texttt{pgamma(4, shape, rate, lower.tail = FALSE)} in R.
\end{recall}

\begin{keyidea}
Counts per month $\Rightarrow$ Poisson likelihood; vague Gamma prior; the answer is a posterior tail area.
\end{keyidea}

\solution
\step{Step 1 --- model.} $Y_t\mid\lambda\iid\dist{Poisson}(\lambda)$, $t=1,\dots,6$; prior $\lambda\sim\dist{Gamma}(0.1,0.1)$ (mean 1, variance 10: vague).
\step{Step 2 --- sufficient statistics.} $\sum Y_t=4+5+2+7+6+6=30$, $n=6$.
\step{Step 3 --- posterior.} $\lambda\mid\mathbf Y\sim\dist{Gamma}(0.1+30,\ 0.1+6)=\dist{Gamma}(30.1,\ 6.1)$: mean $\frac{30.1}{6.1}=4.93$, 95\% interval $(3.33,\ 6.85)$.
\begin{center}\includegraphics[width=0.68\linewidth]{ex04_injuries.pdf}\end{center}
\step{Step 4 --- tail probability} (shaded area): $\Prob(\lambda>4\mid\mathbf Y)=0.853$.
\step{Step 5 --- sensitivity to the prior.}
\begin{center}
\begin{tabular}{lccc}\toprule
Prior & Posterior & Mean & $\Prob(\lambda>4\mid\mathbf Y)$\\\midrule
$\dist{Gamma}(0.01,0.01)$ & $\dist{Gamma}(30.01,6.01)$ & 4.99 & 0.867\\
$\dist{Gamma}(0.1,0.1)$ & $\dist{Gamma}(30.1,6.1)$ & 4.93 & 0.853\\
$\dist{Gamma}(1,1)$ & $\dist{Gamma}(31,7)$ & 4.43 & 0.690\\\bottomrule
\end{tabular}
\end{center}

\begin{sanity}
The sample mean is $30/6=5$ injuries per month, and the posterior mean (4.93) sits right next to it, as it should with a vague prior. \checkmark
\end{sanity}

\begin{answer}
$\lambda\mid\mathbf Y\sim\dist{Gamma}(30.1,6.1)$, posterior mean $\approx4.9$ injuries/month, and $\Prob(\lambda>4\mid\mathbf Y)\approx\mathbf{0.85}$: the factory is probably missing its target.
\end{answer}

\begin{takeaway}
With only six months, the prior matters: $\dist{Gamma}(1,1)$ acts like an extra month with one injury and lowers the probability to 0.69. Always report the prior.
\end{takeaway}

\exercise{2.5 --- Negative binomial likelihood with a beta prior}

\begin{problem}{5}
Assume that $Y|\theta\sim\dist{NegBinomial}(\theta,m)$ (see Appendix A.1) and $\theta\sim\dist{Beta}(a,b)$.
\begin{enumerate}[(a)]
\item Derive the posterior of $\theta$.
\item Plot the posterior of $\theta$, and give its 95\% credible interval assuming $m=5$, $Y=10$, and $a=b=1$.
\end{enumerate}
\end{problem}

\begin{restate}
$Y$ counts the failures before the $m$-th success. With a beta prior on the success probability, find the posterior and evaluate it for $m=5$, $Y=10$.
\end{restate}

\begin{recall}
Appendix A.1: $\Prob(Y=y\mid\theta)=\binom{y+m-1}{y}\theta^m(1-\theta)^y$, $y=0,1,2,\dots$ (failures before the $m$-th success). Beta kernel: $\theta^{a-1}(1-\theta)^{b-1}$.
\end{recall}

\begin{keyidea}
As a function of $\theta$, the likelihood $\theta^m(1-\theta)^Y$ is a Beta kernel, so the Beta prior is conjugate.
\end{keyidea}

\solution
\step{(a) Step 1 --- drop the constant.} The binomial coefficient does not involve $\theta$: $f(Y\mid\theta)\propto\theta^m(1-\theta)^Y$.
\step{(a) Step 2 --- multiply by the prior.}
\[
p(\theta\mid Y)\propto\theta^m(1-\theta)^Y\cdot\theta^{a-1}(1-\theta)^{b-1}=\theta^{(a+m)-1}(1-\theta)^{(b+Y)-1}
\ \Rightarrow\ \theta\mid Y\sim\dist{Beta}(a+m,\ b+Y).
\]
\step{(b) Numbers.} $\dist{Beta}(1+5,\ 1+10)=\dist{Beta}(6,11)$: mean $\frac{6}{17}=0.353$, 95\% interval (\texttt{qbeta(c(.025,.975),6,11)}) $=(0.152,\ 0.587)$ (shaded).
\begin{center}\includegraphics[width=0.68\linewidth]{ex05_negbin.pdf}\end{center}

\begin{sanity}
5 successes and 10 failures: the raw success rate is $5/15=0.333$, close to the posterior mean 0.353 (pulled slightly toward 0.5 by the uniform prior). \checkmark
\end{sanity}

\begin{answer}
(a) $\theta\mid Y\sim\dist{Beta}(a+m,\,b+Y)$. \quad (b) $\dist{Beta}(6,11)$; 95\% credible interval $(0.152,\ 0.587)$.
\end{answer}

\begin{takeaway}
This is the same posterior as 5 successes and 10 failures under a \emph{binomial} likelihood: the stopping rule does not affect Bayesian inference (the \textbf{likelihood principle}).
\end{takeaway}

\exercise{2.6 --- Heart rates and regression to the mean}

\begin{problem}{6}
The distribution of true heart rate across a population is Gaussian with mean 64 beats per minute (bpm) and standard deviation 10 bpm. An individual measurement is unbiased for the true heart rate, but has error standard deviation 5 bpm.
\begin{enumerate}[(a)]
\item Given that an individual has a (single) measured heart rate of $y$ bpm, what is the posterior distribution of their true heart rate? Plot the posterior mean as a function of the observed value and comment on the regression to the mean.
\item Consider a study designed to test a new method to treat high heart rate. To obtain a sample, they measure randomly-selected individuals from the population and include them in the study if and only if their heart rate exceeds 90 bpm. For these individuals, they apply the treatment and measure their heart rate again. The estimate of the treatment effect is the difference in mean heart rate before and after the treatment for the individuals in the study. Assuming the new method has no actual effect on heart rate, compute (using mathematical derivations or simulations) the expected difference in the mean.
\item How could the study in (b) be redesigned to avoid regression to the mean bias?
\end{enumerate}
\end{problem}

\begin{restate}
A noisy heart-rate reading should be pulled back toward the population average. Show this, and show how it creates a fake ``treatment effect'' in a study that only enrols people with high readings.
\end{restate}

\begin{recall}
\begin{itemize}
\item Normal--normal: prior $\theta\sim\dist N(m,\tau^2)$, data $y\mid\theta\sim\dist N(\theta,s^2)$ $\Rightarrow$ posterior precision $\frac1{\tau^2}+\frac1{s^2}$ and posterior mean $w\,y+(1-w)\,m$ with $w=\frac{\tau^2}{\tau^2+s^2}$.
\item \textbf{Truncated normal mean}: if $X\sim\dist N(\mu,\sigma^2)$ then $\E(X\mid X>c)=\mu+\sigma\frac{\phi(z)}{1-\Phi(z)}$ with $z=\frac{c-\mu}{\sigma}$.
\end{itemize}
\end{recall}

\begin{keyidea}
A high reading is partly a high true rate and partly lucky measurement error. The error does not repeat, so the second reading tends to be lower even with no treatment.
\end{keyidea}

\solution
\step{(a) Step 1 --- precisions.} Prior precision $\frac1{100}=0.01$; measurement precision $\frac1{25}=0.04$; total $0.05$ $\Rightarrow$ posterior variance $\frac1{0.05}=20$ (SD 4.47).
\step{(a) Step 2 --- posterior mean.}
\[
\E(\theta\mid y)=\frac{0.01\times64+0.04\times y}{0.05}=0.2\times64+0.8\,y=12.8+0.8y.
\]
So $\theta\mid y\sim\dist{Normal}(12.8+0.8y,\ 20)$.
\begin{center}\includegraphics[width=0.55\linewidth]{ex06_shrinkage.pdf}\end{center}
The line has slope 0.8 (flatter than the dashed diagonal) and crosses it at 64. A reading of 100 implies a true rate of $12.8+80=92.8$; a reading of 40 implies $44.8$. Extreme readings are pulled toward 64: \emph{regression to the mean}.

\step{(b) Step 1 --- set up.}
Let $Y_1=\theta+e_1$ (screening = ``before'') and $Y_2=\theta+e_2$ (``after''; the treatment does nothing). Since $e_2$ has mean 0 and is independent of everything else,
\[
\E(Y_1-Y_2\mid Y_1>90)=\E(Y_1\mid Y_1>90)-\E(\theta\mid Y_1>90).
\]
\step{(b) Step 2 --- use part (a).}
$\E(\theta\mid Y_1)=12.8+0.8Y_1$, so $\E(\theta\mid Y_1>90)=12.8+0.8\,\E(Y_1\mid Y_1>90)$ and
\[
\E(Y_1-Y_2\mid Y_1>90)=0.2\,\E(Y_1\mid Y_1>90)-12.8=0.2\big(\E(Y_1\mid Y_1>90)-64\big).
\]
\step{(b) Step 3 --- truncated normal mean.}
Marginally $Y_1\sim\dist N(64,\ 10^2+5^2=125)$, SD $11.18$. With $z=\frac{90-64}{11.18}=2.326$: $\phi(z)=0.0267$, $1-\Phi(z)=0.0100$, so
\[
\E(Y_1\mid Y_1>90)=64+11.18\times\frac{0.0267}{0.0100}=93.79.
\]
Expected difference $=0.2\times(93.79-64)=\mathbf{5.96}$ bpm.

\begin{sanity}
Simulation: $2\times10^6$ people, 20{,}047 enrolled, average before-minus-after $=5.92$ bpm with a useless treatment. \checkmark
\end{sanity}

\step{(c) Redesign.}
\begin{itemize}
\item \textbf{Randomise}: enrol people with high readings, then randomly give half a placebo. Both groups regress to the mean equally; compare the groups' \emph{after} values.
\item Or take a \emph{fresh baseline} measurement after enrolment (independent of the screening value) and measure change from that.
\item Averaging several screening readings reduces (but does not remove) the bias.
\end{itemize}

\begin{answer}
(a) $\theta\mid y\sim\dist{Normal}(12.8+0.8y,\ 20)$. \quad (b) $\approx\mathbf{6}$ bpm apparent drop with a useless treatment. \quad (c) Use a randomised control group (or a fresh baseline).
\end{answer}

\begin{takeaway}
Selecting people on an extreme noisy value guarantees their next value is, on average, less extreme. Many ``miracle cures'' are regression to the mean.
\end{takeaway}

\exercise{2.7 --- A prior for California wildfires}

\begin{problem}{7}
Over the past 50 years California has experienced an average of $\lambda_0=75$ large wildfires per year. For the next 10 years you will record the number of large fires in California and then fit a Poisson/gamma model to these data. Let the rate of large fires in this future period, $\lambda$, have prior $\lambda\sim\dist{Gamma}(a,b)$. Select $a$ and $b$ so that the prior is uninformative with prior variance around 100 and gives prior probability approximately $\Prob(\lambda>\lambda_0)=0.5$ so that the prior places equal probability on both hypotheses in the test for a change in the rate.
\end{problem}

\begin{restate}
Choose a Gamma prior for the future fire rate whose median is 75 (equal odds of increase or decrease) and whose variance is 100.
\end{restate}

\begin{recall}
$\dist{Gamma}(a,b)$: mean $a/b$, variance $a/b^2$. Its median is slightly \emph{below} its mean (right skew); R gives it as \texttt{qgamma(0.5, a, b)}.
\end{recall}

\begin{keyidea}
Two conditions, two unknowns. The variance condition gives $b$ in terms of $a$; then solve the median condition numerically.
\end{keyidea}

\solution
\step{Step 1 --- quick approximation using the mean.}
Set mean $=75$ and variance $=100$: $\frac ab=75$, $\frac a{b^2}=100$. Dividing, $b=\frac{75}{100}=0.75$ and $a=75\times0.75=56.25$. But the median of $\dist{Gamma}(56.25,0.75)$ is a little below 75: $\Prob(\lambda>75)=0.482$.
\step{Step 2 --- exact solution.}
From $\frac a{b^2}=100$: $b=\sqrt a/10$. Find $a$ such that \texttt{qgamma(0.5, a, sqrt(a)/10)} $=75$ (\texttt{uniroot}): $a=56.91$, $b=0.754$.
\step{Step 3 --- is it ``uninformative''?}
In the Poisson--gamma update, $b$ acts like a number of prior \emph{years} of data. Here $b=0.75$ --- less than one year --- while we will collect 10 years. So the data will dominate.

\begin{sanity}
$\dist{Gamma}(56.91,0.754)$: mean $56.91/0.754=75.4$, SD $\sqrt{56.91}/0.754=10.0$, $\Prob(\lambda>75)=0.500$. \checkmark
\end{sanity}

\begin{answer}
$\lambda\sim\dist{Gamma}(a=56.9,\ b=0.754)$: variance 100 (SD 10), median 75. The simpler $\dist{Gamma}(56.25,0.75)$ (mean 75) gives $\Prob(\lambda>75)=0.48$, close enough to ``approximately 0.5''.
\end{answer}

\begin{takeaway}
Judge ``uninformative'' \emph{relative to the data}: a prior worth 0.75 years is weak against 10 years of counts, even though an SD of 10 fires is not vague in absolute terms.
\end{takeaway}

\exercise{2.8 --- Is the measurement error too large?}

\begin{problem}{8}
An assembly line relies on accurate measurements from an image-recognition algorithm at the first stage of the process. It is known that the algorithm is unbiased, so assume that measurements follow a normal distribution with mean zero, $Y_i|\sigma^2\iid\dist{Normal}(0,\sigma^2)$. Some errors are permissible, but if $\sigma$ exceeds the threshold $c$, then the algorithm must be replaced. You make $n=20$ measurements and observe $\sum_{i=1}^nY_i=-2$ and $\sum_{i=1}^nY_i^2=15$ and conduct a Bayesian analysis with $\dist{InvGamma}(a,b)$ prior. Compute the posterior probability that $\sigma>c$ for:
\begin{enumerate}[(a)]
\item $c=1$ and $a=b=0.1$
\item $c=1$ and $a=b=1.0$
\item $c=2$ and $a=b=0.1$
\item $c=2$ and $a=b=1.0$
\end{enumerate}
For each $c$, compute the ratio of probabilities for the two priors (i.e. $a=b=0.1$ and $a=b=1.0$). Which, if any, of the results are sensitive to the prior?
\end{problem}

\begin{restate}
Estimate the error SD from 20 measurements with known mean 0, and compute the probability it exceeds a threshold under two priors.
\end{restate}

\begin{recall}
\begin{itemize}
\item $\dist{InvGamma}(a,b)$ kernel in $x$: $x^{-a-1}e^{-b/x}$. If $\sigma^2\sim\dist{InvGamma}(A,B)$ then $1/\sigma^2\sim\dist{Gamma}(A,B)$.
\item $\sigma>c\iff\sigma^2>c^2\iff1/\sigma^2<1/c^2$ (all quantities positive).
\end{itemize}
\end{recall}

\begin{keyidea}
Known mean, unknown variance $\Rightarrow$ inverse-gamma posterior. Convert the question about $\sigma$ into a question about the precision $1/\sigma^2$, which is Gamma, and use \texttt{pgamma}.
\end{keyidea}

\solution
\step{Step 1 --- likelihood.}
\[
f(\mathbf Y\mid\sigma^2)=\prod_{i=1}^{20}\frac{1}{\sqrt{2\pi\sigma^2}}e^{-Y_i^2/(2\sigma^2)}\propto(\sigma^2)^{-10}\exp\Big\{-\frac{\sum Y_i^2}{2\sigma^2}\Big\}=(\sigma^2)^{-10}e^{-7.5/\sigma^2}.
\]
\why{$(\sigma^2)^{-1/2}$ twenty times gives $(\sigma^2)^{-10}$; $\sum Y_i^2/2=15/2=7.5$.}
$\sum Y_i=-2$ is \emph{not needed}: the mean is known to be 0, so only $\sum Y_i^2$ enters.
\step{Step 2 --- posterior.}
\[
p(\sigma^2\mid\mathbf Y)\propto(\sigma^2)^{-10}e^{-7.5/\sigma^2}\cdot(\sigma^2)^{-a-1}e^{-b/\sigma^2}=(\sigma^2)^{-(a+10)-1}e^{-(b+7.5)/\sigma^2},
\]
so $\sigma^2\mid\mathbf Y\sim\dist{InvGamma}(a+10,\ b+7.5)$.
\step{Step 3 --- the probability.}
$\Prob(\sigma>c\mid\mathbf Y)=\Prob(1/\sigma^2<1/c^2\mid\mathbf Y)=$ \texttt{pgamma(1/c\^{}2, a + 10, b + 7.5)}.
\begin{center}
\begin{tabular}{c|cc|c}\toprule
 & $a=b=0.1$ & $a=b=1$ & ratio\\\midrule
$c=1$ & 0.2250 & 0.2366 & 0.95\\
$c=2$ & $2.56\times10^{-5}$ & $1.45\times10^{-5}$ & 1.77\\\bottomrule
\end{tabular}
\end{center}

\begin{sanity}
The sample estimate $\hat\sigma^2=\frac{15}{20}=0.75$ ($\hat\sigma=0.87$) is below 1, so $\Prob(\sigma>1)\approx0.23$ (less than half) and $\Prob(\sigma>2)$ tiny both make sense. \checkmark
\end{sanity}

\begin{answer}
(a) 0.225 \quad (b) 0.237 \quad (c) $2.6\times10^{-5}$ \quad (d) $1.4\times10^{-5}$.\\
For $c=1$: \textbf{not} sensitive (ratio 0.95). For $c=2$: \textbf{relatively} sensitive (ratio 1.77) in the extreme tail, but both probabilities are negligible, so the \emph{decision} (keep the algorithm) is the same.
\end{answer}

\begin{takeaway}
Tail probabilities far from the data are always the most prior-sensitive quantities. Judge sensitivity by whether it changes the decision.
\end{takeaway}

\exercise{2.9 --- Induced prior in logistic regression}

\begin{problem}{9}
In simple logistic regression, the probability of a success is regressed onto a single covariate $X$ as
\[
p(X)=\frac{\exp(\alpha+X\beta)}{1+\exp(\alpha+X\beta)}.\qquad(2.51)
\]
Assuming the covariate is distributed as $X\sim\dist{Normal}(0,1)$ and the parameters have priors $\alpha\sim\dist{Normal}(0,c^2)$ and $\beta\sim\dist{Normal}(0,c^2)$, find a prior standard deviation $c$ so that the induced prior on the success probability $p(X)$ is roughly $\dist{Uniform}(0,1)$. That is, for a given $c$ generate $S$ (say $S=1{,}000{,}000$) samples $(X^{(s)},\alpha^{(s)},\beta^{(s)})$ for $s=1,\dots,S$, compute the $S$ success probabilities, and make a histogram of the $S$ probabilities. Repeat this for several $c$ until the histogram is approximately uniform. Report the final value of $c$ and the resulting histogram. Would you call this an uninformative prior for the regression coefficients $\alpha$ and $\beta$?
\end{problem}

\begin{restate}
A prior on the regression coefficients implies a prior on the probability $p(X)$. Find the prior SD $c$ that makes this implied prior flat.
\end{restate}

\begin{recall}
\begin{itemize}
\item If $p\sim\dist{Uniform}(0,1)$, then $\mathrm{logit}(p)=\log\frac{p}{1-p}$ has the standard logistic distribution, variance $\pi^2/3\approx3.29$.
\item For independent $\alpha,\beta,X$ with means 0: $\V(\alpha+X\beta)=\V(\alpha)+\E(X^2)\V(\beta)=c^2+c^2=2c^2$.
\end{itemize}
\end{recall}

\begin{keyidea}
First guess from matching variances, $2c^2\approx3.29\Rightarrow c\approx1.28$; then refine by simulation, measuring flatness with the Kolmogorov--Smirnov (KS) distance to the uniform.
\end{keyidea}

\solution
\step{Step 1 --- simulate the induced prior for a given $c$.}
\begin{lstlisting}
induced_p <- function(c, S) plogis(rnorm(S, 0, c) + rnorm(S) * rnorm(S, 0, c))
\end{lstlisting}
(\texttt{plogis} is the inverse-logit $\frac{e^\eta}{1+e^\eta}$; the three \texttt{rnorm} calls draw $\alpha$, $X$ and $\beta$.)
\step{Step 2 --- search over $c$.}
For $c=0.50,0.55,\dots,3.00$ compute the KS distance $\max_u|\hat F(u)-u|$ between the simulated probabilities and the uniform. The smallest distance is at $c=1.30$; with $S=10^6$ the KS distance is 0.005 (essentially flat).
\begin{center}\includegraphics[width=0.98\linewidth]{ex09_induced.pdf}\end{center}
Left: $c=0.5$ piles up near 0.5. Middle: $c=1.3$ is flat. Right: the ``vague'' $c=10$ pushes almost all mass to $p\approx0$ or $p\approx1$.
\step{Step 3 --- uninformative for $\alpha,\beta$?}
No. With $c=1.3$, $\Prob(|\beta|>2.6)=\Prob(|Z|>2)\approx0.05$: odds ratios above $e^{2.6}\approx13$ per SD of $X$ are considered unlikely. That is a real (though sensible) restriction on the coefficients.

\begin{sanity}
The simulated best value $c=1.30$ agrees with the back-of-envelope variance match $c\approx1.28$. \checkmark
\end{sanity}

\begin{answer}
$c\approx\mathbf{1.3}$ gives an approximately $\dist{Uniform}(0,1)$ induced prior on $p(X)$. It is \textbf{not} uninformative for $\alpha$ and $\beta$; it is mildly informative on the coefficient scale but flat on the probability scale.
\end{answer}

\begin{takeaway}
``Uninformative'' depends on the scale. A prior that looks vague for coefficients ($c=10$) is very \emph{informative} about probabilities. Always check what a prior implies on the scale you care about.
\end{takeaway}

\exercise{2.10 --- Adverse side effects at the Mayo Clinic}

\begin{problem}{10}
The Mayo Clinic conducted a study of $n=50$ patients followed for one year on a new medication and found that 30 patients experienced no adverse side effects (ASE), 12 experienced one ASE, 6 experienced two ASEs and 2 experienced ten ASEs.
\begin{enumerate}[(a)]
\item Derive the posterior of $\lambda$ for the Poisson/gamma model $Y_1,\dots,Y_n|\lambda\iid\dist{Poisson}(\lambda)$ and $\lambda\sim\dist{Gamma}(a,b)$.
\item Use the Poisson/gamma model with $a=b=0.01$ to study the rate of adverse events. Plot the posterior and give the posterior mean and 95\% credible interval.
\item Repeat this analysis with $\dist{Gamma}(0.1,0.1)$ and $\dist{Gamma}(1,1)$ priors and discuss sensitivity to the prior.
\item Plot the data versus the $\dist{Poisson}(\hat\lambda)$ PMF, where $\hat\lambda$ is the posterior mean of $\lambda$ from part (b). Does the Poisson likelihood fit well?
\item The current medication is thought to have around one adverse side effect per year. What is the posterior probability that this new medication has a higher side effect rate than the previous medication? Are the results sensitive to the prior?
\end{enumerate}
\end{problem}

\begin{restate}
Estimate the yearly rate of side effects from 50 patients, check prior sensitivity and model fit, and compare with a rate of 1 per year.
\end{restate}

\begin{recall}
For a Poisson distribution, the variance equals the mean. If the data's variance is much larger than its mean, the data are \textbf{over-dispersed} and a Poisson model is too narrow.
\end{recall}

\begin{keyidea}
Poisson--gamma conjugacy for (a)--(c) and (e); for (d) compare observed frequencies with $\dist{Poisson}(\hat\lambda)$ probabilities.
\end{keyidea}

\solution
\step{(a) Derivation.}
\[
p(\lambda\mid\mathbf Y)\propto\prod_{i=1}^n\frac{e^{-\lambda}\lambda^{Y_i}}{Y_i!}\cdot\lambda^{a-1}e^{-b\lambda}\propto\lambda^{a+\sum Y_i-1}e^{-(b+n)\lambda}
\ \Rightarrow\ \lambda\mid\mathbf Y\sim\dist{Gamma}\Big(a+\sum Y_i,\ b+n\Big).
\]
\step{(b) Numbers.} $\sum Y_i=0\times30+1\times12+2\times6+10\times2=44$, $n=50$: $\lambda\mid\mathbf Y\sim\dist{Gamma}(44.01,\ 50.01)$, mean $0.880$, 95\% interval $(0.639,\ 1.158)$.

\step{(c) Sensitivity.}
\begin{center}
\begin{tabular}{lcccc}\toprule
Prior & Posterior & Mean & 95\% interval & $\Prob(\lambda>1)$\\\midrule
$\dist{Gamma}(0.01,0.01)$ & $\dist{Gamma}(44.01,50.01)$ & 0.880 & (0.639, 1.158) & 0.180\\
$\dist{Gamma}(0.1,0.1)$ & $\dist{Gamma}(44.1,50.1)$ & 0.880 & (0.640, 1.158) & 0.180\\
$\dist{Gamma}(1,1)$ & $\dist{Gamma}(45,51)$ & 0.882 & (0.644, 1.158) & 0.182\\\bottomrule

The fit is \textbf{poor}. $\dist{Poisson}(0.88)$ predicts $50\times e^{-0.88}=20.7$ patients with no events (observed: 30), and $\Prob(Y\ge10)=3.5\times10^{-8}$, yet 2 of 50 patients had ten events. The sample variance is 4.03, about 4.6 times the mean 0.88: strong \textbf{over-dispersion}.

\begin{answer}
(a) $\dist{Gamma}(a+\sum Y_i,\ b+n)$. \quad (b) $\dist{Gamma}(44.01,50.01)$: mean 0.88, 95\% CI $(0.64,1.16)$. \quad (c) Insensitive.\\

\exercise{2.11 --- Days until a smoker relapses}

\begin{problem}{11}
A smoker is trying to quit and their goal is to abstain for one week because studies have shown that this is the most difficult period. Denote $Y$ as the number of days before the smoker relapses, and $\theta$ as the probability that they relapse on any given day of the study.
\begin{enumerate}[(a)]
\item Argue that a negative binomial distribution (Appendix A.1) for $Y|\theta$ is reasonable. What are the most important assumptions being made, and are they valid for this smoking cessation attempt?
\item Assuming a negative binomial likelihood and $\dist{Uniform}(0,1)$ prior for $\theta$, what is the prior probability the smoker will be smoke-free for at least 7 days?
\item Plot the posterior distribution of $\theta$ assuming $Y=5$.
\item Given this attempt resulted in $Y=5$ and that $\theta$ is the same in the second attempt, what is the posterior predictive probability the second attempt will result in at least 7 smoke-free days?
\end{enumerate}
\end{problem}

\begin{restate}
Each day the smoker relapses with probability $\theta$. Model the number of smoke-free days, update with an attempt that lasted 5 days, and predict the next attempt.
\end{restate}

\begin{recall}
\begin{itemize}
\item $\dist{NegBinomial}(\theta,m)$ (Appendix A.1) counts failures before the $m$-th success. With $m=1$ it is the \textbf{geometric}: $\Prob(Y=y\mid\theta)=\theta(1-\theta)^y$.
\item ``At least $k$ failures before the first success'' means the first $k$ trials all failed: $\Prob(Y\ge k\mid\theta)=(1-\theta)^k$.
\item $\int_0^1\theta^{a-1}(1-\theta)^{b-1}\dd\theta=B(a,b)$.
\end{itemize}
\end{recall}

\begin{keyidea}
Each day is a trial; ``relapse'' is the ``success''. Smoke-free days are the failures before the first success.
\end{keyidea}

\solution
\step{(a) Justification and assumptions.}
Counting days until the first event in a sequence of trials is exactly a negative binomial with $m=1$. Assumptions:
\begin{enumerate}[(i)]
\item the relapse probability $\theta$ is the \textbf{same every day} --- doubtful: cravings are worst early on, so the daily risk probably \emph{decreases};
\item days are \textbf{independent} --- doubtful: a stressful day raises the next day's risk too;
\item time is counted in whole days.
\end{enumerate}
Still a reasonable first approximation.

\step{(b) Prior predictive probability.}
\[
\Prob(Y\ge7)=\int_0^1\Prob(Y\ge7\mid\theta)\,\pi(\theta)\dd\theta=\int_0^1(1-\theta)^7\dd\theta=\Big[-\frac{(1-\theta)^8}{8}\Big]_0^1=\frac18=0.125.
\]

\step{(c) Posterior with $Y=5$.}
\[
p(\theta\mid Y=5)\propto\theta(1-\theta)^5\cdot1=\theta^{2-1}(1-\theta)^{6-1}\ \Rightarrow\ \theta\mid Y\sim\dist{Beta}(2,6),
\]
mean $\frac28=0.25$, 95\% interval $(0.037,\ 0.579)$.
\begin{center}\includegraphics[width=0.66\linewidth]{ex11_posterior.pdf}\end{center}

\step{(d) Posterior predictive.}
\[
\Prob(Y^*\ge7\mid Y=5)=\int_0^1(1-\theta)^7\frac{\theta(1-\theta)^5}{B(2,6)}\dd\theta=\frac{B(2,13)}{B(2,6)}.
\]
With $B(a,b)=\frac{(a-1)!(b-1)!}{(a+b-1)!}$:
\[
\frac{B(2,13)}{B(2,6)}=\frac{1!\,12!/14!}{1!\,5!/7!}=\frac{1/(13\cdot14)}{1/(6\cdot7)}=\frac{6\cdot7}{13\cdot14}=\frac{42}{182}=\frac{3}{13}\approx0.231.
\]

\begin{sanity}
Monte Carlo: draw $\theta\sim\dist{Beta}(2,6)$, then $Y^*\sim\dist{Geometric}(\theta)$; the fraction with $Y^*\ge7$ is 0.2307. \checkmark
\end{sanity}

\begin{answer}
(a) Geometric $=\dist{NegBinomial}(\theta,1)$; assumes a constant, independent daily relapse risk (doubtful). \quad (b) $1/8=0.125$. \quad (c) $\dist{Beta}(2,6)$. \quad (d) $3/13\approx\mathbf{0.231}$.
\end{answer}

\begin{takeaway}
Plugging in $\hat\theta=0.25$ would give $0.75^7=0.133$, understating the chance of a good week because it ignores the possibility of a small $\theta$.
\end{takeaway}

\exercise{2.12 --- Censored relapse times}

\begin{problem}{12}
A smoking cessation study followed $n=20$ patients until they relapsed to smoking. The trial terminated after six weeks and so many patients did not relapse during the observation period; i.e., their responses was censored. Seven patients relapsed during the observation period at times 0.5, 0.9, 1.1, 2.4, 3.8, 4.0 and 5.2 weeks, and the rest were censored at 6 weeks. Assume the times to event $Y_1,\dots,Y_n$ follow the survival analysis model $Y_i|\lambda\iid\dist{Exponential}(\lambda)$ with prior $\lambda\sim\dist{Gamma}(a,b)$.
\begin{enumerate}[(a)]
\item Show that the posterior distribution of $\lambda$ is a gamma distribution (hint: the likelihood contribution of a censored observation is $\Prob(Y_i>6|\lambda)=\exp(-6\lambda)$).
\item Compute the posterior mean and 95\% interval for $\lambda$ for several values of $a$ and $b$. Are the results sensitive to the prior?
\end{enumerate}
\end{problem}

\begin{restate}
Seven patients relapsed at known times; thirteen were still smoke-free when the study ended. Use both kinds of information to estimate the relapse rate.
\end{restate}

\begin{recall}
For $\dist{Exponential}(\lambda)$: density $\lambda e^{-\lambda y}$ and survival function $\Prob(Y>t)=e^{-\lambda t}$.
\end{recall}

\begin{keyidea}
An observed event contributes its density; a censored patient contributes the probability of surviving past 6 weeks. The posterior then depends only on the number of events $d$ and the total time at risk $T$.
\end{keyidea}

\solution
\step{(a) Step 1 --- likelihood.}
\[
L(\lambda)=\underbrace{\prod_{i=1}^{7}\lambda e^{-\lambda y_i}}_{\text{7 relapses}}\times\underbrace{\prod_{j=1}^{13}e^{-6\lambda}}_{\text{13 censored}}=\lambda^{7}\exp\Big\{-\lambda\Big(\sum_{i=1}^7y_i+13\times6\Big)\Big\}.
\]
\step{(a) Step 2 --- total time at risk.}
$\sum y_i=0.5+0.9+1.1+2.4+3.8+4.0+5.2=17.9$ and $13\times6=78$, so $T=95.9$ patient-weeks and $L(\lambda)=\lambda^7e^{-95.9\lambda}$.
\step{(a) Step 3 --- posterior.}
\[
p(\lambda\mid\text{data})\propto\lambda^{7}e^{-95.9\lambda}\cdot\lambda^{a-1}e^{-b\lambda}=\lambda^{(a+7)-1}e^{-(b+95.9)\lambda}
\ \Rightarrow\ \lambda\mid\text{data}\sim\dist{Gamma}(a+7,\ b+95.9).
\]
\step{(b) Several priors.}
\begin{center}
\begin{tabular}{lccc}\toprule
Prior & Posterior & Mean (per week) & 95\% interval\\\midrule
$\dist{Gamma}(0.01,0.01)$ & $\dist{Gamma}(7.01,95.91)$ & 0.073 & (0.029, 0.136)\\
$\dist{Gamma}(0.1,0.1)$ & $\dist{Gamma}(7.1,96.0)$ & 0.074 & (0.030, 0.138)\\
$\dist{Gamma}(1,1)$ & $\dist{Gamma}(8,96.9)$ & 0.083 & (0.036, 0.149)\\
$\dist{Gamma}(2,10)$ & $\dist{Gamma}(9,105.9)$ & 0.085 & (0.039, 0.149)\\\bottomrule
\end{tabular}
\end{center}
The vague priors agree; priors that add one or two pseudo-events move the mean by about 15\%, because there are only 7 real events.

\begin{sanity}
Classical estimate: events $\div$ time at risk $=7/95.9=0.073$ per week --- the same as the vague-prior posterior mean. \checkmark
\end{sanity}

\begin{answer}
(a) $\lambda\mid\text{data}\sim\dist{Gamma}(a+7,\ b+95.9)$. \quad (b) Vague priors: mean $\approx0.073$ relapses/week (mean time to relapse $\approx14$ weeks), 95\% CI $\approx(0.03,0.14)$; mildly sensitive because there are only 7 events.
\end{answer}

\begin{takeaway}
\textbf{Never drop censored observations.} Using only the 7 relapse times gives a rate of $1/2.56=0.39$ per week --- five times too high.
\end{takeaway}

\exercise{2.13 --- Mixture-of-experts prior for an alarm}

\begin{problem}{13}
You are designing a very small experiment to determine the sensitivity of a new security alarm system. You will simulate five robbery attempts and record the number of these attempts that trigger the alarm. Because the dataset will be small, you ask two experts for their opinion. One expects the alarm probability to be 0.95 with standard deviation 0.05, and the other expects 0.80 with standard deviation 0.20.
\begin{enumerate}[(a)]
\item Translate these two priors into beta PDFs, and plot the two beta PDFs and the corresponding mixture of experts prior with equal weight given to each expert.
\item Now you conduct the experiment, and the alarm is triggered in every simulated robbery. Plot the posterior of the alarm probability under a uniform prior, each experts' prior, and the mixture of experts prior.
\end{enumerate}
\end{problem}

\begin{restate}
Turn two experts' (mean, SD) opinions into beta priors, average them, and update all priors with 5 alarms out of 5 attempts.
\end{restate}

\begin{recall}
\begin{itemize}
\item \textbf{Beta moment matching}: for mean $\mu$ and variance $v$, $a+b=\frac{\mu(1-\mu)}{v}-1$, $a=\mu(a+b)$, $b=(1-\mu)(a+b)$.
\item \textbf{Marginal likelihood} of $Y$ successes in $n$ trials under $\dist{Beta}(a,b)$: $m(Y)=\binom nY\frac{B(a+Y,\,b+n-Y)}{B(a,b)}$.
\item Mixture update: primer, point 5.
\end{itemize}
\end{recall}

\begin{keyidea}
Each expert's prior updates in the usual conjugate way; the mixture weights update in proportion to how well each expert predicted the data.
\end{keyidea}

\solution
\step{(a) Expert 1} (mean 0.95, SD 0.05, variance 0.0025):
$a+b=\frac{0.95\times0.05}{0.0025}-1=19-1=18$, so $a=0.95\times18=17.1$, $b=0.05\times18=0.9$: $\dist{Beta}(17.1,\ 0.9)$.
\step{(a) Expert 2} (mean 0.80, SD 0.20, variance 0.04):
$a+b=\frac{0.8\times0.2}{0.04}-1=4-1=3$, so $a=2.4$, $b=0.6$: $\dist{Beta}(2.4,\ 0.6)$.
\step{(a) Mixture.} $\pi(\theta)=0.5\,\dist{Beta}(\theta;17.1,0.9)+0.5\,\dist{Beta}(\theta;2.4,0.6)$.
\begin{center}\includegraphics[width=0.72\linewidth]{ex13_priors.pdf}\end{center}

\step{(b) Step 1 --- each conjugate posterior} ($Y=5$, $n=5$, add 5 to $a$, 0 to $b$):
uniform $\to\dist{Beta}(6,1)$ (mean 0.857); expert 1 $\to\dist{Beta}(22.1,0.9)$ (mean 0.961); expert 2 $\to\dist{Beta}(7.4,0.6)$ (mean 0.925).
\step{(b) Step 2 --- updated mixture weights.}
\[
w_1'\propto0.5\,\frac{B(22.1,0.9)}{B(17.1,0.9)},\qquad w_2'\propto0.5\,\frac{B(7.4,0.6)}{B(2.4,0.6)}\quad\Rightarrow\quad w_1'=0.617,\ \ w_2'=0.383.
\]
(The binomial coefficient $\binom55=1$ is common to both and cancels.)
\step{(b) Step 3 --- mixture posterior.} $0.617\,\dist{Beta}(22.1,0.9)+0.383\,\dist{Beta}(7.4,0.6)$, mean $0.617\times0.961+0.383\times0.925=0.947$.
\begin{center}\includegraphics[width=0.72\linewidth]{ex13_posteriors.pdf}\end{center}

\begin{sanity}
Expert 1 predicted 5/5 more confidently (prior mean 0.95), so the data should shift weight toward expert 1 --- and they do (50\% $\to$ 62\%). \checkmark
\end{sanity}

\begin{answer}
(a) Expert 1 $=\dist{Beta}(17.1,0.9)$, expert 2 $=\dist{Beta}(2.4,0.6)$, mixture $=$ their 50/50 average.\\
(b) Posteriors: uniform $\dist{Beta}(6,1)$; expert 1 $\dist{Beta}(22.1,0.9)$; expert 2 $\dist{Beta}(7.4,0.6)$; mixture $0.617\,\dist{Beta}(22.1,0.9)+0.383\,\dist{Beta}(7.4,0.6)$ (mean 0.947).
\end{answer}

\begin{takeaway}
A mixture prior is a principled way to combine disagreeing experts: the data decide whom to trust.
\end{takeaway}

\exercise{2.14 --- Probability or odds: which scale is ``flat''?}

\begin{problem}{14}
Let $\theta$ be the binomial success probability and $\gamma=\theta/(1-\theta)$ be the corresponding odds of a success. Use Monte Carlo simulation (you can discard samples with $\gamma>100$) to explore the effects of parameterization on prior choice in this context.
\begin{enumerate}[(a)]
\item If $\theta\sim\dist{Uniform}(0,1)$, what is the induced prior on $\gamma$?
\item If $\theta\sim\dist{Beta}(0.5,0.5)$, what is the induced prior on $\gamma$?
\item If $\gamma\sim\dist{Uniform}(0,100)$, what is the induced prior on $\theta$?
\item If $\gamma\sim\dist{Gamma}(1,1)$, what is the induced prior on $\theta$?
\item Would you say any of these priors are simultaneously uninformative for both $\theta$ and $\gamma$?
\end{enumerate}
\end{problem}

\begin{restate}
Put a prior on the probability $\theta$ and see what it says about the odds $\gamma$, and vice versa.
\end{restate}

\begin{recall}
\textbf{Change of variables}: if $\gamma=g(\theta)$ is monotone with inverse $\theta=h(\gamma)$, then $f_\gamma(\gamma)=f_\theta\big(h(\gamma)\big)\,|h'(\gamma)|$.
Here $\theta=h(\gamma)=\frac{\gamma}{1+\gamma}$, $h'(\gamma)=\frac{1}{(1+\gamma)^2}$, and $1-\theta=\frac1{1+\gamma}$.
In the other direction $\gamma=\frac\theta{1-\theta}$ and $\frac{\dd\gamma}{\dd\theta}=\frac1{(1-\theta)^2}$.
\end{recall}

\begin{keyidea}
Monte Carlo: draw from the prior on one scale and transform each draw. The change-of-variables formula gives the exact curves to overlay.
\end{keyidea}

\solution
\step{(a) $\theta\sim\dist{Uniform}(0,1)$.} $f_\theta=1$, so $f_\gamma(\gamma)=1\cdot\frac{1}{(1+\gamma)^2}$ for $\gamma>0$. Decreasing, median 1 (since $\theta=0.5\iff\gamma=1$), very heavy tail ($\E\gamma=\infty$).
\step{(b) $\theta\sim\dist{Beta}(0.5,0.5)$.} $f_\theta(\theta)=\frac{1}{\pi}\theta^{-1/2}(1-\theta)^{-1/2}$. Substitute $\theta=\frac{\gamma}{1+\gamma}$, $1-\theta=\frac{1}{1+\gamma}$:
\[
f_\gamma(\gamma)=\frac1\pi\Big(\frac{\gamma}{1+\gamma}\Big)^{-1/2}\Big(\frac{1}{1+\gamma}\Big)^{-1/2}\frac{1}{(1+\gamma)^2}=\frac1\pi\,\gamma^{-1/2}(1+\gamma)\frac{1}{(1+\gamma)^2}=\frac{1}{\pi\sqrt\gamma\,(1+\gamma)}.
\]
An infinite spike at 0 plus a heavy tail.
\step{(c) $\gamma\sim\dist{Uniform}(0,100)$.} $f_\theta(\theta)=\frac1{100}\cdot\frac{1}{(1-\theta)^2}$ for $0<\theta<\frac{100}{101}$. Almost all mass near 1: $\Prob(\theta>0.9)=\Prob(\gamma>9)=\frac{91}{100}=0.91$.
\step{(d) $\gamma\sim\dist{Gamma}(1,1)=\dist{Exponential}(1)$.} $f_\theta(\theta)=e^{-\theta/(1-\theta)}\frac{1}{(1-\theta)^2}$ on $(0,1)$: hump-shaped, falling to 0 near 1 ($\Prob(\theta>0.9)=e^{-9}\approx0.0001$).

Simulated histograms ($10^6$ draws, $\gamma>100$ discarded) with exact densities in red:
\begin{center}\includegraphics[width=0.95\linewidth]{ex14_reparam.pdf}\end{center}

\step{(e)} \textbf{None} is flat on both scales. A flat $\theta$ gives a strongly skewed $\gamma$; a flat $\gamma$ pushes $\theta$ toward 1. Because $\gamma=\theta/(1-\theta)$ is nonlinear, flatness on one scale implies non-flatness on the other.

\begin{sanity}
Each red curve lies on top of its simulated histogram. \checkmark
\end{sanity}

\begin{answer}
(a) $\frac1{(1+\gamma)^2}$ \quad (b) $\frac1{\pi\sqrt\gamma(1+\gamma)}$ \quad (c) $\frac{1}{100(1-\theta)^2}$ on $(0,\frac{100}{101})$ \quad (d) $\frac{e^{-\theta/(1-\theta)}}{(1-\theta)^2}$\\
(e) \textbf{No} prior is uninformative on both scales.
\end{answer}

\begin{takeaway}
This motivates the \textbf{Jeffreys prior}: not flat, but giving the same answer whichever parameterisation you start from. The $\dist{Beta}(0.5,0.5)$ in (b) is the binomial Jeffreys prior.
\end{takeaway}

\exercise{2.15 --- A flat prior can give a proper posterior}

\begin{problem}{15}
Say $Y|\mu\sim\dist{Normal}(\mu,1)$ and $\mu$ has improper prior $\pi(\mu)=1$ for all $\mu$. Prove that the posterior distribution of $\mu$ is a proper PDF.
\end{problem}

\begin{restate}
The flat prior does not integrate to 1; show that the posterior still does.
\end{restate}

\begin{recall}
A posterior is proper iff the marginal $m(Y)=\int f(Y\mid\mu)\pi(\mu)\dd\mu$ is finite. Gaussian integral: $\int_{-\infty}^\infty e^{-(\mu-Y)^2/2}\dd\mu=\sqrt{2\pi}$.
\end{recall}

\begin{keyidea}
As a function of $\mu$, the normal density $f(Y\mid\mu)$ has exactly the shape of a $\dist N(Y,1)$ density, which integrates to 1.
\end{keyidea}

\solution
\step{Step 1 --- marginal.}
\[
m(Y)=\int_{-\infty}^{\infty}\frac{1}{\sqrt{2\pi}}e^{-(Y-\mu)^2/2}\cdot1\dd\mu=\frac{1}{\sqrt{2\pi}}\sqrt{2\pi}=1<\infty.
\]
\why{$(Y-\mu)^2=(\mu-Y)^2$, so the integrand is the $\dist N(Y,1)$ density in the variable $\mu$.}
\step{Step 2 --- posterior.}
\[
p(\mu\mid Y)=\frac{f(Y\mid\mu)\pi(\mu)}{m(Y)}=\frac{1}{\sqrt{2\pi}}e^{-(\mu-Y)^2/2},
\]
non-negative and integrating to 1: a proper density, namely $\dist N(Y,1)$.

\begin{answer}
$m(Y)=1<\infty$, so $p(\mu\mid Y)$ is proper: $\mu\mid Y\sim\dist{Normal}(Y,1)$.
\end{answer}

\begin{takeaway}
An improper prior is acceptable only if the posterior is proper --- this must be checked model by model (Exercises 16--17 show cases where it fails).
\end{takeaway}

\exercise{2.16 --- When an improper prior must fail}

\begin{problem}{16}
Suppose that the likelihood $f(\mathbf Y|\theta)\ge c(\mathbf Y)$ for all $\theta$. Show that an improper prior on $\theta$ will yield an improper posterior.
\end{problem}

\begin{restate}
If the likelihood never drops below some positive level, the data cannot ``tame'' a prior with infinite total mass.
\end{restate}

\begin{recall}
Improper prior: $\pi(\theta)\ge0$ and $\int\pi(\theta)\dd\theta=\infty$. Proper posterior $\iff m(\mathbf Y)=\int f(\mathbf Y\mid\theta)\pi(\theta)\dd\theta<\infty$. If $g\ge h\ge0$ then $\int g\ge\int h$.
\end{recall}

\begin{keyidea}
Bound the marginal likelihood from below by $c(\mathbf Y)$ times the (infinite) prior mass.
\end{keyidea}

\solution
Assume $c(\mathbf Y)>0$ (otherwise the bound says nothing). Since $\pi(\theta)\ge0$, multiplying the inequality $f(\mathbf Y\mid\theta)\ge c(\mathbf Y)$ by $\pi(\theta)$ keeps its direction, and integrating:
\[
m(\mathbf Y)=\int f(\mathbf Y\mid\theta)\pi(\theta)\dd\theta\ \ge\ \int c(\mathbf Y)\pi(\theta)\dd\theta=c(\mathbf Y)\int\pi(\theta)\dd\theta=c(\mathbf Y)\times\infty=\infty.
\]
So the normalising constant is infinite: $f(\mathbf Y\mid\theta)\pi(\theta)$ cannot be scaled into a density, i.e.\ the posterior is improper.

\begin{answer}
$m(\mathbf Y)\ge c(\mathbf Y)\int\pi(\theta)\dd\theta=\infty$ whenever $c(\mathbf Y)>0$, so the posterior is improper.
\end{answer}

\begin{takeaway}
For a proper posterior under an improper prior, the likelihood must \emph{decay} in every direction where the prior has infinite mass.
\end{takeaway}

\exercise{2.17 --- Mixture models and improper priors}

\begin{problem}{17}
Consider a sample of size $n$ from the mixture of two normal densities given by $f(y|\theta)=0.5\phi(y)+0.5\phi(y-\theta)$ where $\phi(z)$ denotes the standard normal density function. Show that any improper prior for $\theta$ leads to an improper posterior.
\end{problem}

\begin{restate}
Half the data come from $\dist N(0,1)$ and half from $\dist N(\theta,1)$. Show that no improper prior on $\theta$ can work.
\end{restate}

\begin{recall}
Exercise 16: a likelihood bounded below by a positive constant $c(\mathbf y)$ makes every improper prior give an improper posterior. $\phi(z)>0$ for every $z$.
\end{recall}

\begin{keyidea}
Find a lower bound for the likelihood that does not involve $\theta$.
\end{keyidea}

\solution
\step{Step 1 --- one observation.} Since $\phi(y_i-\theta)\ge0$,
\[
f(y_i\mid\theta)=0.5\phi(y_i)+0.5\phi(y_i-\theta)\ \ge\ 0.5\phi(y_i)\quad\text{for every }\theta.
\]
\step{Step 2 --- the whole sample.} The observations are independent, so the likelihood is a product of positive numbers each bounded below:
\[
f(\mathbf y\mid\theta)=\prod_{i=1}^n f(y_i\mid\theta)\ \ge\ \prod_{i=1}^n0.5\phi(y_i)=:c(\mathbf y)>0.
\]
\step{Step 3 --- conclude.} By Exercise 16, $m(\mathbf y)\ge c(\mathbf y)\int\pi(\theta)\dd\theta=\infty$.

\begin{answer}
$f(\mathbf y\mid\theta)\ge\prod_i0.5\phi(y_i)>0$ for all $\theta$, so by Exercise 16 every improper prior gives an improper posterior.
\end{answer}

\begin{takeaway}
Intuition: as $\theta\to\pm\infty$ the likelihood does not vanish --- every observation could have come from the first component, which says nothing about $\theta$. Mixture models need proper priors (see Chapter 3, Exercise 15).
\end{takeaway}

\exercise{2.19 --- Jeffreys prior for the Poisson}

\begin{problem}{19}
Say $Y|\lambda\sim\dist{Poisson}(\lambda)$.
\begin{enumerate}[(a)]
\item Derive and plot the Jeffreys prior for $\lambda$.
\item Is this prior proper?
\item Derive the posterior, and give conditions on $Y$ to ensure it is proper.
\end{enumerate}
\end{problem}

\begin{restate}
Compute the Fisher information of the Poisson, turn it into the Jeffreys prior, and check whether prior and posterior are proper.
\end{restate}

\begin{recall}
Jeffreys prior: $\pi(\lambda)\propto\sqrt{I(\lambda)}$, $I(\lambda)=-\E\Big[\frac{\partial^2}{\partial\lambda^2}\log f(Y\mid\lambda)\Big]$. For a Poisson, $\E Y=\lambda$. A Gamma$(a,b)$ is a proper distribution iff $a>0$ and $b>0$.
\end{recall}

\begin{keyidea}
Differentiate the log-likelihood twice, take the expectation over $Y$, then take a square root.
\end{keyidea}

\solution
\step{(a) Step 1 --- log-likelihood.} $\log f(Y\mid\lambda)=Y\log\lambda-\lambda-\log Y!$.
\step{(a) Step 2 --- derivatives.} $\frac{\partial}{\partial\lambda}\log f=\frac{Y}{\lambda}-1$; \ $\frac{\partial^2}{\partial\lambda^2}\log f=-\frac{Y}{\lambda^2}$.
\step{(a) Step 3 --- Fisher information.} $I(\lambda)=-\E\big(-\frac{Y}{\lambda^2}\big)=\frac{\E Y}{\lambda^2}=\frac{\lambda}{\lambda^2}=\frac1\lambda$.
\step{(a) Step 4 --- Jeffreys prior.} $\pi(\lambda)\propto\sqrt{1/\lambda}=\lambda^{-1/2}$, $\lambda>0$.
\begin{center}\includegraphics[width=0.85\linewidth]{ex19_20_jeffreys.pdf}\end{center}
\step{(b) Proper?} $\int_0^\infty\lambda^{-1/2}\dd\lambda=\big[2\sqrt\lambda\big]_0^\infty=\infty$. \textbf{Improper} (the upper tail is the problem).
\step{(c) Posterior.}
\[
p(\lambda\mid Y)\propto\lambda^{Y}e^{-\lambda}\cdot\lambda^{-1/2}=\lambda^{(Y+\frac12)-1}e^{-\lambda}\ \Rightarrow\ \lambda\mid Y\sim\dist{Gamma}\big(Y+\tfrac12,\ 1\big).
\]
Shape $Y+\frac12>0$ and rate $1>0$ for every $Y\in\{0,1,2,\dots\}$, so the posterior is always proper.

\begin{answer}
(a) $\pi(\lambda)\propto\lambda^{-1/2}$. \quad (b) Improper. \quad (c) $\lambda\mid Y\sim\dist{Gamma}(Y+\frac12,1)$, proper for \textbf{every} $Y\ge0$, including $Y=0$.
\end{answer}

\begin{takeaway}
The Jeffreys prior is the improper ``$\dist{Gamma}(\frac12,0)$''; it is the default prior used in Exercise 21.
\end{takeaway}

\exercise{2.20 --- Jeffreys prior for the exponential rate}

\begin{problem}{20}
Say $Y|\lambda\sim\dist{Gamma}(1,\lambda)$.
\begin{enumerate}[(a)]
\item Derive and plot the Jeffreys prior for $\lambda$.
\item Is this prior proper?
\item Derive the posterior, and give conditions on $Y$ to ensure it is proper.
\end{enumerate}
\end{problem}

\begin{restate}
Same as Exercise 19, for an exponential waiting time with rate $\lambda$.
\end{restate}

\begin{recall}
$\dist{Gamma}(1,\lambda)$ is the exponential with rate $\lambda$: $f(y\mid\lambda)=\lambda e^{-\lambda y}$, $y>0$.
\end{recall}

\begin{keyidea}
Same recipe as Exercise 19. Here the second derivative does not even involve $Y$, so no expectation is needed.
\end{keyidea}

\solution
\step{(a) Step 1 --- log-likelihood.} $\log f=\log\lambda-\lambda Y$.
\step{(a) Step 2 --- derivatives.} $\frac{\partial}{\partial\lambda}\log f=\frac1\lambda-Y$; \ $\frac{\partial^2}{\partial\lambda^2}\log f=-\frac{1}{\lambda^2}$.
\step{(a) Step 3 --- Jeffreys prior.} $I(\lambda)=\frac1{\lambda^2}\Rightarrow\pi(\lambda)\propto\frac1\lambda$ (right panel of the figure in Exercise 19).
\step{(b) Proper?} $\int_0^\infty\lambda^{-1}\dd\lambda=\big[\log\lambda\big]_0^\infty$ diverges at \emph{both} ends. \textbf{Improper.}
\step{(c) Posterior.}
\[
p(\lambda\mid Y)\propto\lambda e^{-\lambda Y}\cdot\lambda^{-1}=e^{-\lambda Y}\ \Rightarrow\ \lambda\mid Y\sim\dist{Gamma}(1,\ Y)=\dist{Exponential}(Y).
\]
Proper iff the rate $Y>0$. Since $Y$ is a positive continuous variable, $Y>0$ with probability 1. (If $Y=0$, $p\propto1$ would be improper.)

\begin{answer}
(a) $\pi(\lambda)\propto1/\lambda$. \quad (b) Improper. \quad (c) $\lambda\mid Y\sim\dist{Gamma}(1,Y)$, proper iff $Y>0$.
\end{answer}

\begin{takeaway}
$1/\lambda$ is the classic scale-invariant prior (flat on $\log\lambda$), the Jeffreys prior for any rate or scale parameter.
\end{takeaway}

\begin{enumerate}[(a)]
\item Analyze the data in each county separately using the Jeffreys prior distribution, and report the posterior 95\% credible set for each county.
\item Let $\hat p_i$ be the sample proportion of commuters in county $i$ that approve of the HOV lane (e.g., $\hat p_1=12/(12+50)=0.194$). Select $a$ and $b$ so that the mean and variance of the $\dist{Beta}(a,b)$ distribution match the mean and variance of the sample proportions $\hat p_1,\dots,\hat p_{10}$.
\item Conduct an empirical Bayesian analysis by computing the 95\% posterior credible sets that results from analyzing each county separately using the $\dist{Beta}(a,b)$ prior you computed in (b).
\item How do the results from (a) and (c) differ? What are the advantages and disadvantages of these two analyses?
\end{enumerate}
\end{problem}

\begin{restate}
Estimate approval in each county two ways --- alone with an objective prior, and with a prior estimated from all counties together --- and compare.
\end{restate}

\begin{recall}
\begin{itemize}
\item Binomial Jeffreys prior: $\dist{Beta}(\frac12,\frac12)$.
\item Beta moment matching (Exercise 13): $a+b=\frac{m(1-m)}{v}-1$, $a=m(a+b)$, $b=(1-m)(a+b)$.
\end{itemize}
\end{recall}

\begin{keyidea}
Empirical Bayes = estimate the prior from the spread of the county proportions, then use it in every county. Small counties get pulled toward the overall average (\textbf{shrinkage}).
\end{keyidea}

\solution
\step{(a) Jeffreys analysis.}
Each county separately: $\theta_i\mid Y_i\sim\dist{Beta}(Y_i+\frac12,\ n_i-Y_i+\frac12)$. E.g.\ county 1: $\dist{Beta}(12.5,\ 50.5)$. The 95\% intervals are in the left half of the table below.

\step{(b) The empirical prior.}
Mean of the ten $\hat p_i$: $m=0.480$; variance: $v=0.02026$. Then
\[
a+b=\frac{0.480\times0.520}{0.02026}-1=11.32,\qquad a=0.480\times11.32=5.43,\qquad b=0.520\times11.32=5.89.
\]
So the prior is $\dist{Beta}(5.43,\ 5.89)$: mean 0.48, worth about 11 respondents.

\step{(c) Empirical Bayes analysis.}
Each county with the $\dist{Beta}(5.43,5.89)$ prior: $\theta_i\mid Y_i\sim\dist{Beta}(5.43+Y_i,\ 5.89+n_i-Y_i)$ (right half of the table).

\why{EB posterior mean $=\frac{5.43+Y_i}{11.32+n_i}$: a weighted average of the prior mean 0.48 (weight $\frac{11.32}{11.32+n_i}$) and $\hat p_i$. Small $n_i$ $\Rightarrow$ more weight on 0.48.}

\step{(d) Comparison.}
\begin{itemize}
\item \textbf{Shrinkage}: EB pulls estimates toward 0.48, most for small counties (county 4, $n=15$: $0.344\to0.396$; county 6, $n=23$: $0.646\to0.595$), hardly at all for large ones (county 2, $n=240$).
\item \textbf{Precision}: EB intervals are narrower (average width 0.216 vs.\ 0.237).
\item \textbf{Jeffreys} --- pros: objective, each county stands alone, no assumption that counties are similar. Cons: ignores information from other counties; wide intervals for small counties.
\item \textbf{Empirical Bayes} --- pros: borrows strength, better estimates on average for small counties. Cons: uses the data twice and treats $a,b$ as known, so its intervals are somewhat too narrow; assumes counties are exchangeable.
\end{itemize}

\begin{answer}
(a) Jeffreys intervals: see table. \quad (b) $a=5.43$, $b=5.89$. \quad (c) EB intervals: see table.\\
(d) EB shrinks small counties toward 0.48 and narrows intervals, at the cost of ignoring uncertainty in $(a,b)$. The fully Bayesian fix is a \emph{hierarchical model} with priors on $a,b$ (Section 2.2 / Chapter 7).
\end{answer}

\begin{takeaway}
Empirical Bayes sits between ``each county alone'' and ``all counties identical''; the data decide the amount of shrinkage (here weak: about 11 pseudo-respondents).
\end{takeaway}
